# Scheduling-Ships
Christian Navarro, Jorge Gutiérrez, Mauricio Luna

Scheduling Ships - Calendarizadores iniciales

Compilar:
    make

Ejecutar Round Robin:
    ./scheduling_ships rr 2
    make run-rr

Ejecutar Prioridad:
    ./scheduling_ships priority
    make run-priority

Ejecutar SJF:
    ./scheduling_ships sjf
    make run-sjf

Ejecutar STRN:
    ./scheduling_ships strn
    make run-strn

Ejecutar FCFS:
    ./scheduling_ships fcfs
    make run-fcfs
